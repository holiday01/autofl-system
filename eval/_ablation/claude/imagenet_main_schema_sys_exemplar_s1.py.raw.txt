class Summary(Enum):

    AVERAGE = 1
    SUM = 2

class AverageMeter(object):
    """Computes and stores the average and current value"""
    def __init__(self, name, use_accel, fmt=':f', summary_type=Summary.AVERAGE):
        self.name = name
        self.use_accel = use_accel
        self.fmt = fmt
        self.summary_type = summary_type
        self.reset()

    def reset(self):
        self.val = 0
        self.avg = 0
        self.sum = 0
        self.count = 0

    def update(self, val, n=1):
        self.val = val
        self.sum += val * n
        self.count += n
        self.avg = self.sum / self.count

    def all_reduce(self):
        if self.use_accel:
            device = torch.accelerator.current_accelerator()
        else:
            device = torch.device("cpu")
        total = torch.tensor([self.sum, self.count], dtype=torch.float32, device=device)
        dist.all_reduce(total, dist.ReduceOp.SUM, async_op=False)
        self.sum, self.count = total.tolist()
        self.avg = self.sum / self.count

    def __str__(self):
        fmtstr = '{name} {val' + self.fmt + '} ({avg' + self.fmt + '})'
        return fmtstr.format(**self.__dict__)

    def summary(self):
        fmtstr = ''
        if self.summary_type is Summary.NONE:
            fmtstr = ''
        elif self.summary_type is Summary.AVERAGE:
            fmtstr = '{name} {avg:.3f}'
        elif self.summary_type is Summary.SUM:
            fmtstr = '{name} {sum:.3f}'
        elif self.summary_type is Summary.COUNT:
            fmtstr = '{name} {count:.3f}'
        else:
            raise ValueError('invalid summary type %r' % self.summary_type)
        return fmtstr.format(**self.__dict__)


class ProgressMeter(object):
    def __init__(self, num_batches, meters, prefix=""):
        self.batch_fmtstr = self._get_batch_fmtstr(num_batches)
        self.meters = meters
        self.prefix = prefix

    def display(self, batch):
        entries = [self.prefix + self.batch_fmtstr.format(batch)]
        entries += [str(meter) for meter in self.meters]
        print('\t'.join(entries))

    def display_summary(self):
        entries = [" *"]
        entries += [meter.summary() for meter in self.meters]
        print(' '.join(entries))

    def _get_batch_fmtstr(self, num_batches):
        num_digits = len(str(num_batches // 1))
        fmt = '{:' + str(num_digits) + 'd}'
        return '[' + fmt + '/' + fmt.format(num_batches) + ']'


def accuracy(output, target, topk=(1,)):
    """Computes the accuracy over the k top predictions for the specified values of k"""
    with torch.no_grad():
        maxk = max(topk)
        batch_size = target.size(0)
        _, pred = output.topk(maxk, 1, True, True)
        pred = pred.t()
        correct = pred.eq(target.view(1, -1).expand_as(pred))
        res = []
        for k in topk:
            correct_k = correct[:k].reshape(-1).float().sum(0, keepdim=True)
            res.append(correct_k.mul_(100.0 / batch_size))
        return res

class _SyntheticImageNetDataset(torch.utils.data.Dataset):
    """Tiny in-memory fake ImageNet dataset for smoke-testing only."""

def build_model(config: dict) -> nn.Module:
    """
    Instantiate a torchvision model.

    Relevant config keys:
      arch          (str)  – torchvision model name, default "resnet18"
      pretrained    (bool) – load DEFAULT ImageNet weights, default False
      model_kwargs  (dict) – extra kwargs forwarded to the model constructor
                             (e.g. {"num_classes": 10} for fine-tuning)
    """
    arch = config.get("arch", "resnet18")
    pretrained = config.get("pretrained", False)
    kwargs = dict(config.get("model_kwargs", {}))

    if arch not in models.__dict__ or not callable(models.__dict__[arch]):
        raise ValueError(
            f"Unknown torchvision architecture '{arch}'. "
            f"Available: {sorted(n for n in models.__dict__ if n.islower() and not n.startswith('__') and callable(models.__dict__[n]))}"
        )

    if pretrained and "weights" not in kwargs:
        # Use the modern weights API; fall back gracefully for older torchvision.
        try:
            kwargs["weights"] = "DEFAULT"
            model = models.__dict__[arch](**kwargs)
        except TypeError:
            kwargs.pop("weights", None)
            kwargs["pretrained"] = True
            model = models.__dict__[arch](**kwargs)
    else:
        model = models.__dict__[arch](**kwargs)

    return model


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    """
    Return a DataLoader for the requested split ("train" or "val").

    Relevant config keys:
      data_path             (str)   – root directory; expects ImageFolder layout
                                      or a 'train/' subdirectory, default "."
      local.batch_size      (int)   – default 16
      local.num_workers     (int)   – default 4
      local.pin_memory      (bool)  – default True
      val_ratio             (float) – fraction of data used for validation,
                                      default 0.1
      seed                  (int)   – random_split seed, default 42
      synthetic_n           (int)   – number of synthetic samples, default 200
      model_kwargs.num_classes (int)– number of classes for synthetic labels,
                                      default 1000
      allow_synthetic_data  (bool)  – MUST be True to use synthetic fallback;
                                      if False and real data is absent,
                                      raises FileNotFoundError
    """
    local = config.get("local", {})
    batch_size  = local.get("batch_size",  config.get("batch_size",  16))
    num_workers = local.get("num_workers", config.get("num_workers", 4))
    pin_memory  = local.get("pin_memory",  True)

    data_path = config.get("data_path", ".")
    val_ratio = config.get("val_ratio", 0.1)
    seed      = config.get("seed", 42)

    normalize = transforms.Normalize(
        mean=[0.485, 0.456, 0.406],
        std=[0.229, 0.224, 0.225],
    )
    train_transform = transforms.Compose([
        transforms.RandomResizedCrop(224),
        transforms.RandomHorizontalFlip(),
        transforms.ToTensor(),
        normalize,
    ])
    val_transform = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        normalize,
    ])

    # Resolve dataset root: prefer data_path/train/ if it exists so that a
    # full ImageNet tree (data_path/{train,val}/) works out of the box.
    candidate_roots = [
        os.path.join(data_path, "train"),
        data_path,
    ]
    dataset_root = next(
        (r for r in candidate_roots if os.path.isdir(r)),
        None,
    )

    full_dataset: torch.utils.data.Dataset | None = None
    if dataset_root is not None:
        # Load once with the transform appropriate for the requested split so
        # that augmentation is applied correctly after random_split.
        transform = train_transform if split == "train" else val_transform
        try:
            ds = datasets.ImageFolder(dataset_root, transform=transform)
            if len(ds) > 0:
                full_dataset = ds
        except (FileNotFoundError, RuntimeError):
            full_dataset = None

    if full_dataset is None:
        if not config.get("allow_synthetic_data", False):
            raise FileNotFoundError(
                f"No ImageFolder-compatible dataset found at '{data_path}' "
                f"(also tried '{os.path.join(data_path, 'train')}'). "
                "Set config['allow_synthetic_data'] = True to use a synthetic "
                "fallback for smoke-testing."
            )
        num_classes = config.get("model_kwargs", {}).get("num_classes", 1000)
        n_synthetic  = config.get("synthetic_n", 200)
        full_dataset = _SyntheticImageNetDataset(n=n_synthetic, num_classes=num_classes)

    n_val   = max(1, int(len(full_dataset) * val_ratio))
    n_train = len(full_dataset) - n_val
    train_ds, val_ds = random_split(
        full_dataset,
        [n_train, n_val],
        generator=torch.Generator().manual_seed(seed),
    )

    ds = train_ds if split == "train" else val_ds
    return DataLoader(
        ds,
        batch_size=batch_size,
        shuffle=(split == "train"),
        num_workers=num_workers,
        pin_memory=pin_memory and torch.cuda.is_available(),
        drop_last=(split == "train"),
    )


def train_step(
    model: nn.Module,
    batch: tuple | list | dict,
    optimizer,
    config: dict,
) -> torch.Tensor:
    """
    ONE forward pass.  Returns the raw loss tensor WITH grad_fn attached.

    The FL runtime calls loss.backward() and optimizer.step() externally —
    do NOT do either here, and do NOT detach() or .item() the returned loss.
    """
    device = next(model.parameters()).device

    if isinstance(batch, (list, tuple)):
        batch = [b.to(device) if isinstance(b, torch.Tensor) else b for b in batch]
        inputs, targets = batch[0], batch[1]
    elif isinstance(batch, dict):
        batch = {
            k: v.to(device) if isinstance(v, torch.Tensor) else v
            for k, v in batch.items()
        }
        inputs  = batch.get("input",  batch.get("x",      batch.get("image")))
        targets = batch.get("label",  batch.get("y",      batch.get("target")))
    else:
        raise TypeError(f"Unsupported batch type: {type(batch)}")

    outputs = model(inputs)
    criterion = nn.CrossEntropyLoss()
    loss = criterion(outputs, targets)
    # grad_fn is attached; the FL runtime owns backward() and optimizer.step().
    return loss
