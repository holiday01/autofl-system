```python
from os import path
from typing import Optional

import torch
from torch.nn import functional as F
from torch.utils.data import DataLoader, random_split

from lightning.pytorch import LightningDataModule, LightningModule, cli_lightning_logo
from lightning.pytorch.cli import LightningCLI
from lightning.pytorch.demos.mnist_datamodule import MNIST
from lightning.pytorch.utilities.imports import _TORCHVISION_AVAILABLE

if _TORCHVISION_AVAILABLE:
    from torchvision import transforms

DATASETS_PATH = path.join(path.dirname(__file__), "..", "..", "Datasets")


class Backbone(torch.nn.Module):
    """
    >>> Backbone()  # doctest: +ELLIPSIS +NORMALIZE_WHITESPACE
    Backbone(
      (l1): Linear(...)
      (l2): Linear(...)
    )
    """

    def __init__(self, hidden_dim=128):
        super().__init__()
        self.l1 = torch.nn.Linear(28 * 28, hidden_dim)
        self.l2 = torch.nn.Linear(hidden_dim, 10)

    def forward(self, x):
        x = x.view(x.size(0), -1)
        x = torch.relu(self.l1(x))
        return torch.relu(self.l2(x))


# ---------------------------------------------------------------------------
# FL client API
# ---------------------------------------------------------------------------

def build_model(config: dict) -> torch.nn.Module:
    """Instantiate and return the Backbone model.

    Args:
        config: FL config dict.  Reads ``config.get("model_kwargs", {})``
                and forwards the contents as constructor keyword arguments
                (e.g. ``{"hidden_dim": 256}``).

    Returns:
        An initialised :class:`Backbone` instance.
    """
    model_kwargs = config.get("model_kwargs", {})
    return Backbone(**model_kwargs)


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    """Return a DataLoader for the requested split.

    A 90 / 10 train / val split is produced with :func:`random_split` from
    the MNIST *train* partition (mirroring the original script's approach).

    Args:
        config: FL config dict.  Relevant keys:

            * ``config["local"]["batch_size"]`` – batch size (default 16).
            * ``config["data_path"]`` – root directory for MNIST (default ".").
            * ``config["allow_synthetic_data"]`` – **must** be ``True`` for
              the synthetic fallback to activate; if ``False`` (the default)
              and real data is unavailable, :exc:`FileNotFoundError` is raised.

        split: ``"train"`` or ``"val"``.

    Returns:
        A :class:`~torch.utils.data.DataLoader` for the requested split.

    Raises:
        ValueError: If *split* is not ``"train"`` or ``"val"``.
        FileNotFoundError: If real data is unavailable and
            ``allow_synthetic_data`` is ``False``.
    """
    if split not in ("train", "val"):
        raise ValueError(f"Unknown split: {split!r}. Expected 'train' or 'val'.")

    batch_size = config.get("local", {}).get("batch_size", 16)
    data_path = config.get("data_path", ".")

    # ------------------------------------------------------------------ #
    # Attempt to load the real MNIST dataset                               #
    # ------------------------------------------------------------------ #
    _real_data_exc = None
    try:
        transform = transforms.ToTensor() if _TORCHVISION_AVAILABLE else None
        full_dataset = MNIST(data_path, train=True, download=False, transform=transform)

        total = len(full_dataset)
        val_size = max(1, int(total * 0.1))
        train_size = total - val_size

        train_subset, val_subset = random_split(
            full_dataset,
            [train_size, val_size],
            generator=torch.Generator().manual_seed(42),
        )

        chosen = train_subset if split == "train" else val_subset
        return DataLoader(chosen, batch_size=batch_size, shuffle=(split == "train"))

    except (FileNotFoundError, RuntimeError) as exc:
        _real_data_exc = exc

    # ------------------------------------------------------------------ #
    # Real data unavailable – synthetic fallback (opt-in only)            #
    # ------------------------------------------------------------------ #
    if not config.get("allow_synthetic_data", False):
        raise FileNotFoundError(
            f"MNIST dataset not found at '{data_path}'. "
            "Download it first or set config['allow_synthetic_data'] = True "
            "to use randomly generated tensors instead."
        ) from _real_data_exc

    n_samples = 500 if split == "train" else 100
    images = torch.randn(n_samples, 1, 28, 28)
    labels = torch.randint(0, 10, (n_samples,))
    synthetic_ds = torch.utils.data.TensorDataset(images, labels)
    return DataLoader(synthetic_ds, batch_size=batch_size, shuffle=(split == "train"))


def train_step(
    model: torch.nn.Module,
    batch,
    optimizer: torch.optim.Optimizer,
    config: dict,
) -> torch.Tensor:
    """Run a single forward pass and return the loss tensor.

    The FL runtime is responsible for calling ``loss.backward()`` and
    ``optimizer.step()``; this function must **not** do either.

    Args:
        model:     The model returned by :func:`build_model`.
        batch:     A ``(images, labels)`` tuple from :func:`build_dataloader`.
        optimizer: Provided by the FL runtime (unused here, kept for API
                   consistency).
        config:    FL config dict (currently unused inside this function).

    Returns:
        Scalar cross-entropy loss tensor **with gradients attached**.
    """
    device = next(model.parameters()).device
    x, y = batch
    x = x.to(device)
    y = y.to(device)

    y_hat = model(x)
    loss = F.cross_entropy(y_hat, y)
    return loss  # backward() and optimizer.step() are handled by the FL runtime
```

---

**What changed and why**

| Concern | Decision |
|---|---|
| **LightningModule stripped** | `LitClassifier` is a wrapper; the actual neural network is `Backbone`. The FL API exposes `Backbone` directly, preserving its architecture byte-for-byte. |
| **Loss function** | Taken verbatim from `LitClassifier.training_step`: `F.cross_entropy(y_hat, y)`. |
| **Data split** | `random_split` with a fixed seed (42, matching the original) creates a 90/10 train/val partition from MNIST's 60 000-sample train set. |
| **Synthetic fallback** | Gated strictly on `config.get("allow_synthetic_data", False)`. If the flag is absent/`False` and MNIST is missing, `FileNotFoundError` is raised immediately — no silent synthetic training. |
| **`train_step` contract** | Only one forward pass; `loss.backward()` and `optimizer.step()` are intentionally omitted so the FL runtime controls aggregation timing. |
| **All original imports kept** | `LightningDataModule`, `LightningModule`, `LightningCLI`, `cli_lightning_logo`, `MNIST`, etc. are retained as required by the rules. |
