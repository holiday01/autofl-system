```python
import os
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import DataLoader, random_split, TensorDataset
from torchvision import datasets, transforms


# ── Model: PyTorch port of the mini-Xception network ─────────────────────────
# Original used Keras/TF; architecture is preserved exactly.
# Input is expected in [0, 255] float range; rescaling to [0, 1] happens
# inside the model (matching Keras Rescaling(1./255)).

class SeparableConv2d(nn.Module):
    """Depthwise-separable convolution — equivalent to Keras SeparableConv2D."""

    def __init__(self, in_channels: int, out_channels: int,
                 kernel_size: int, padding: int = 0, bias: bool = True):
        super().__init__()
        self.depthwise = nn.Conv2d(
            in_channels, in_channels, kernel_size,
            padding=padding, groups=in_channels, bias=False,
        )
        self.pointwise = nn.Conv2d(in_channels, out_channels, 1, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.pointwise(self.depthwise(x))


class MiniXception(nn.Module):
    """
    PyTorch port of the mini-Xception built in the Keras Cats-vs-Dogs example.

    Architecture (preserved from original):
        Rescaling → Conv2d(128,3,s=2) → BN → ReLU
        → 3× residual block [256, 512, 728] with SeparableConv2d + MaxPool
        → SeparableConv2d(1024,3) → BN → ReLU
        → GlobalAvgPool → Dropout(0.25)
        → Linear(1 if binary else num_classes)
    """

    def __init__(self, input_shape: tuple = (3, 180, 180), num_classes: int = 2):
        super().__init__()
        in_channels = input_shape[0]

        # Entry block
        self.entry_conv = nn.Conv2d(in_channels, 128, 3, stride=2, padding=1, bias=False)
        self.entry_bn   = nn.BatchNorm2d(128)

        # Residual blocks
        block_sizes = [256, 512, 728]
        prev         = 128
        self.res_blocks = nn.ModuleList()
        self.res_skips  = nn.ModuleList()

        for size in block_sizes:
            self.res_blocks.append(nn.Sequential(
                nn.ReLU(),
                SeparableConv2d(prev, size, 3, padding=1, bias=False),
                nn.BatchNorm2d(size),
                nn.ReLU(),
                SeparableConv2d(size, size, 3, padding=1, bias=False),
                nn.BatchNorm2d(size),
                nn.MaxPool2d(3, stride=2, padding=1),
            ))
            self.res_skips.append(
                nn.Conv2d(prev, size, 1, stride=2, padding=0, bias=False)
            )
            prev = size

        # Top block
        self.top_sep = SeparableConv2d(728, 1024, 3, padding=1, bias=False)
        self.top_bn  = nn.BatchNorm2d(1024)

        # Classification head
        self.gap     = nn.AdaptiveAvgPool2d(1)
        self.dropout = nn.Dropout(0.25)
        out_units    = 1 if num_classes == 2 else num_classes
        self.fc      = nn.Linear(1024, out_units)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Rescaling: [0, 255] → [0, 1]  (matches Keras Rescaling(1./255))
        x = x / 255.0

        # Entry block
        x = F.relu(self.entry_bn(self.entry_conv(x)))

        # Residual blocks
        for block, skip in zip(self.res_blocks, self.res_skips):
            x = block(x) + skip(x)

        # Top block
        x = F.relu(self.top_bn(self.top_sep(x)))

        # Head
        x = self.gap(x).flatten(1)
        x = self.dropout(x)
        return self.fc(x)


# ── FL interface ──────────────────────────────────────────────────────────────

def build_model(config: dict) -> nn.Module:
    """
    Instantiate and return MiniXception.

    config keys used:
        model_kwargs (dict): passed as **kwargs to MiniXception.__init__
            e.g. {"input_shape": (3, 180, 180), "num_classes": 2}
    """
    kwargs = config.get("model_kwargs", {})
    return MiniXception(**kwargs)


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    """
    Return a DataLoader for 'train' or 'val'.

    config keys used:
        data_path (str)         : root directory for ImageFolder
                                  (expects Cat/ and Dog/ sub-dirs, i.e. PetImages/)
        local.batch_size (int)  : default 16
        image_size (tuple)      : (H, W), default (180, 180)
        num_workers (int)       : default 2
        allow_synthetic_data (bool): MUST be True to use the synthetic fallback

    Raises FileNotFoundError if data_path is missing and allow_synthetic_data is False.
    """
    batch_size  = config.get("local", {}).get("batch_size", 16)
    data_path   = config.get("data_path", ".")
    image_size  = config.get("image_size", (180, 180))
    num_workers = config.get("num_workers", 2)

    data_available = os.path.isdir(data_path)

    if not data_available:
        if not config.get("allow_synthetic_data", False):
            raise FileNotFoundError(
                f"Data directory '{data_path}' not found. "
                "Provide a valid config['data_path'] pointing to the PetImages/ folder, "
                "or set config['allow_synthetic_data'] = True to use synthetic tensors "
                "for smoke-testing only."
            )
        # ── Synthetic fallback (only reached when allow_synthetic_data=True) ──
        n_total = 200
        n_train = int(0.8 * n_total)
        n_val   = n_total - n_train
        c, h, w = 3, image_size[0], image_size[1]
        # Images in [0, 255] float range to match model expectations
        images  = torch.randint(0, 256, (n_total, c, h, w), dtype=torch.float32)
        labels  = torch.randint(0, 2,   (n_total,),          dtype=torch.float32)
        full_ds = TensorDataset(images, labels)
        train_ds, val_ds = random_split(
            full_ds, [n_train, n_val],
            generator=torch.Generator().manual_seed(1337),
        )
    else:
        # ToTensor gives [0,1]; multiply by 255 restores the [0,255] range the
        # model's internal Rescaling layer expects.
        tf = transforms.Compose([
            transforms.Resize(image_size),
            transforms.ToTensor(),
            transforms.Lambda(lambda t: t * 255.0),
        ])
        full_ds = datasets.ImageFolder(data_path, transform=tf)
        n_total = len(full_ds)
        n_train = int(0.8 * n_total)
        n_val   = n_total - n_train
        train_ds, val_ds = random_split(
            full_ds, [n_train, n_val],
            generator=torch.Generator().manual_seed(1337),
        )

    chosen = train_ds if split == "train" else val_ds
    return DataLoader(
        chosen,
        batch_size=batch_size,
        shuffle=(split == "train"),
        num_workers=num_workers,
        pin_memory=True,
    )


def train_step(
    model: nn.Module,
    batch,
    optimizer: torch.optim.Optimizer,
    config: dict,
) -> torch.Tensor:
    """
    Run ONE forward pass and return the loss tensor with grad attached.

    The FL runtime calls loss.backward() and optimizer.step(); this function
    must NOT do either.

    Supports:
        Binary classification  (model output shape [B, 1]) → BCEWithLogitsLoss
        Multi-class            (model output shape [B, C]) → CrossEntropyLoss
    """
    images, labels = batch
    device = next(model.parameters()).device
    images = images.to(device)
    labels = labels.to(device)

    logits = model(images)

    if logits.shape[-1] == 1:
        # Binary path — matches original BinaryCrossentropy(from_logits=True)
        logits  = logits.squeeze(-1)
        labels  = labels.float()
        loss_fn = nn.BCEWithLogitsLoss()
    else:
        # Multi-class path
        labels  = labels.long()
        loss_fn = nn.CrossEntropyLoss()

    return loss_fn(logits, labels)
```

### What was converted and why

| Original (Keras/TF) | FL module (PyTorch) |
|---|---|
| `keras.Model` with functional API | `MiniXception(nn.Module)` — identical topology |
| `layers.SeparableConv2D` | `SeparableConv2d` — depthwise + pointwise `nn.Conv2d` |
| `layers.Rescaling(1./255)` | `x = x / 255.0` at the top of `forward()` |
| `layers.GlobalAveragePooling2D` | `nn.AdaptiveAvgPool2d(1)` |
| `keras.utils.image_dataset_from_directory` | `torchvision.datasets.ImageFolder` + `random_split` |
| `BinaryCrossentropy(from_logits=True)` | `nn.BCEWithLogitsLoss` |
| `model.fit(...)` | `train_step` — forward only; no `.backward()` / `.step()` |

Key FL rules enforced:
- **`train_step`** returns the loss tensor with grad but never calls `backward()` or `optimizer.step()`.
- **Synthetic fallback** is strictly gated on `config.get("allow_synthetic_data", False)`; a missing data path raises `FileNotFoundError` by default.
- **`build_dataloader`** produces deterministic 80/20 `random_split` subsets for both `"train"` and `"val"`.
