```python
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import DataLoader, TensorDataset, random_split

# ---------------------------------------------------------------------------
# Original script constants (kept for reference)
# ---------------------------------------------------------------------------
num_classes = 10
input_shape = (28, 28, 1)  # Keras channels-last; PyTorch uses (1, 28, 28)


# ---------------------------------------------------------------------------
# Model: direct PyTorch equivalent of the original Keras Sequential
#   Conv2D(32,3,relu) → MaxPool(2) → Conv2D(64,3,relu) → MaxPool(2)
#   → Flatten → Dropout(0.5) → Dense(10, softmax)
#
# Spatial walk on 28×28 input (no padding):
#   conv1(3×3) → 26×26  |  pool(2×2) → 13×13
#   conv2(3×3) → 11×11  |  pool(2×2) →  5×5
#   flatten → 64 × 5 × 5 = 1600
# ---------------------------------------------------------------------------
class MNISTConvNet(nn.Module):
    def __init__(self, num_classes: int = 10, dropout: float = 0.5):
        super().__init__()
        self.conv1   = nn.Conv2d(1,  32, kernel_size=3)
        self.pool1   = nn.MaxPool2d(2, 2)
        self.conv2   = nn.Conv2d(32, 64, kernel_size=3)
        self.pool2   = nn.MaxPool2d(2, 2)
        self.dropout = nn.Dropout(p=dropout)
        self.fc      = nn.Linear(64 * 5 * 5, num_classes)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = F.relu(self.conv1(x))
        x = self.pool1(x)
        x = F.relu(self.conv2(x))
        x = self.pool2(x)
        x = torch.flatten(x, 1)
        x = self.dropout(x)
        return self.fc(x)          # raw logits; CrossEntropyLoss expects logits


# ---------------------------------------------------------------------------
# FL interface
# ---------------------------------------------------------------------------

def build_model(config: dict) -> nn.Module:
    """Instantiate and return the MNISTConvNet.

    Supported model_kwargs:
        num_classes (int, default 10)
        dropout     (float, default 0.5)
    """
    kwargs = config.get("model_kwargs", {})
    return MNISTConvNet(**kwargs)


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    """Return a DataLoader for the requested split ("train" or "val").

    Data source priority
    --------------------
    1. torchvision MNIST downloaded to config["data_path"]  (no auto-download)
    2. Synthetic tensors — ONLY when config["allow_synthetic_data"] is True;
       raises FileNotFoundError otherwise.

    The full training set is split 90 / 10 into train / val via random_split.
    """
    batch_size = config.get("local", {}).get("batch_size", 16)
    data_path  = config.get("data_path", ".")

    full_dataset = None

    # ── attempt to load the real MNIST dataset ──────────────────────────────
    try:
        from torchvision import datasets, transforms

        transform = transforms.Compose([
            transforms.ToTensor(),        # uint8 HWC → float32 CHW in [0, 1]
        ])
        full_dataset = datasets.MNIST(
            root=data_path,
            train=True,
            download=False,
            transform=transform,
        )
    except Exception:
        pass   # handled below

    # ── fallback / error ─────────────────────────────────────────────────────
    if full_dataset is None:
        if not config.get("allow_synthetic_data", False):
            raise FileNotFoundError(
                f"MNIST dataset not found at '{data_path}'. "
                "Either download it first (torchvision MNIST, download=True) "
                "or set config['allow_synthetic_data'] = True to proceed with "
                "random synthetic tensors (for smoke-tests only)."
            )
        # Synthetic: 1 000 samples of shape (1, 28, 28), integer labels
        _n = 1000
        _x = torch.randn(_n, 1, 28, 28)
        _y = torch.randint(0, num_classes, (_n,))
        full_dataset = TensorDataset(_x, _y)

    # ── train / val split ────────────────────────────────────────────────────
    total     = len(full_dataset)
    val_size  = max(1, int(0.1 * total))
    train_size = total - val_size
    train_ds, val_ds = random_split(
        full_dataset,
        [train_size, val_size],
        generator=torch.Generator().manual_seed(42),
    )

    chosen  = train_ds if split == "train" else val_ds
    shuffle = split == "train"
    return DataLoader(chosen, batch_size=batch_size, shuffle=shuffle)


def train_step(
    model: nn.Module,
    batch,
    optimizer: torch.optim.Optimizer,
    config: dict,
) -> torch.Tensor:
    """Run one forward pass and return the loss tensor (grad attached).

    The FL runtime is responsible for loss.backward() and optimizer.step().
    This function does NOT call either.
    """
    device = next(model.parameters()).device

    images, labels = batch
    images = images.to(device)          # (B, 1, 28, 28), float32 in [0, 1]
    labels = labels.to(device)          # (B,) int64 class indices

    logits = model(images)              # (B, num_classes)
    loss   = F.cross_entropy(logits, labels)   # equivalent to categorical_crossentropy
    return loss                         # grad is attached; caller handles backward
```

---

**Key conversion decisions:**

| Original (Keras) | FL client (PyTorch) |
|---|---|
| `keras.Sequential` with TF-style layers | `MNISTConvNet(nn.Module)` — identical topology |
| Channels-last `(28,28,1)` input | Channels-first `(1,28,28)` — same spatial math, same flattened size (1600) |
| `softmax` output + `categorical_crossentropy` | Raw logits + `F.cross_entropy` — numerically identical, numerically stabler |
| One-hot `y` via `to_categorical` | Integer class indices — required by `CrossEntropyLoss` |
| `keras.datasets.mnist.load_data()` | `torchvision.datasets.MNIST(download=False)` with 90/10 `random_split` |
| `model.fit(…, validation_split=0.1)` | `random_split` seeded at 42, same 10 % val fraction |
| `model.compile / fit` handles backward | `train_step` returns loss only; no `backward()` / `step()` called |
