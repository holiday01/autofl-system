"""
Auto-generated FL client module.
Original script: an example DCGAN from PyTorch examples.

This FL client module is designed to federate the training of the Discriminator
component of a DCGAN. The Generator model is assumed to be either fixed or
provided by the FL server for a given round, as the FL client contract
is designed for a single model and optimizer per train_step.

Exposes:
  build_model(config)               -> nn.Module

CONTRACT (read carefully before copying this pattern):
  - train_step performs ONE forward pass and returns the raw loss tensor.
  - The returned tensor MUST have grad_fn attached (do NOT call .detach()).
  - Do NOT call loss.backward() inside train_step.
  - Do NOT call optimizer.step() or optimizer.zero_grad() inside train_step.
  - Do NOT call .item() on the returned loss.
  The FL runtime owns backward(), step(), and metric extraction.

import random
import torch
import torch.nn as nn

import torch.utils.data

import torchvision.transforms as transforms

# Helper to map dataset names to channel info and normalization
_DATASET_INFO = {
    'cifar10': {'nc': 3, 'mean': (0.5, 0.5, 0.5), 'std': (0.5, 0.5, 0.5)},
    'lsun': {'nc': 3, 'mean': (0.5, 0.5, 0.5), 'std': (0.5, 0.5, 0.5)},
    'imagenet': {'nc': 3, 'mean': (0.5, 0.5, 0.5), 'std': (0.5, 0.5, 0.5)},

    'mnist': {'nc': 1, 'mean': (0.5,), 'std': (0.5,)},
}

def _get_dataset_params(dataset_name):
    return _DATASET_INFO.get(dataset_name.lower(), {'nc': 3, 'mean': (0.5, 0.5, 0.5), 'std': (0.5, 0.5, 0.5)})


# custom weights initialization called on netG and netD
def weights_init(m):
    classname = m.__class__.__name__
    if classname.find('Conv') != -1:
        torch.nn.init.normal_(m.weight, 0.0, 0.02)
    elif classname.find('BatchNorm') != -1:
        torch.nn.init.normal_(m.weight, 1.0, 0.02)
        torch.nn.init.zeros_(m.bias)


class Generator(nn.Module):
    def __init__(self, nz=100, ngf=64, nc=3):
        super(Generator, self).__init__()
        self.main = nn.Sequential(
            # input is Z, going into a convolution
            nn.ConvTranspose2d(     nz, ngf * 8, 4, 1, 0, bias=False),
            nn.BatchNorm2d(ngf * 8),
            nn.ReLU(True),
            # state size. (ngf*8) x 4 x 4
            nn.ConvTranspose2d(ngf * 8, ngf * 4, 4, 2, 1, bias=False),
            nn.BatchNorm2d(ngf * 4),
            nn.ReLU(True),
            # state size. (ngf*4) x 8 x 8
            nn.ConvTranspose2d(ngf * 4, ngf * 2, 4, 2, 1, bias=False),
            nn.BatchNorm2d(ngf * 2),
            nn.ReLU(True),
            # state size. (ngf*2) x 16 x 16
            nn.ConvTranspose2d(ngf * 2,     ngf, 4, 2, 1, bias=False),
            nn.BatchNorm2d(ngf),
            nn.ReLU(True),
            # state size. (ngf) x 32 x 32
            nn.ConvTranspose2d(    ngf,      nc, 4, 2, 1, bias=False),
            nn.Tanh()
            # state size. (nc) x 64 x 64
        )

    def forward(self, input):
        # Data parallelization is handled by the FL framework, not inside the module
        return self.main(input)


class Discriminator(nn.Module):
    def __init__(self, ndf=64, nc=3):
        super(Discriminator, self).__init__()
        self.main = nn.Sequential(
            # input is (nc) x 64 x 64
            nn.Conv2d(nc, ndf, 4, 2, 1, bias=False),
            nn.LeakyReLU(0.2, inplace=True),
            # state size. (ndf) x 32 x 32
            nn.Conv2d(ndf, ndf * 2, 4, 2, 1, bias=False),
            nn.BatchNorm2d(ndf * 2),
            nn.LeakyReLU(0.2, inplace=True),
            # state size. (ndf*2) x 16 x 16
            nn.Conv2d(ndf * 2, ndf * 4, 4, 2, 1, bias=False),
            nn.BatchNorm2d(ndf * 4),
            nn.LeakyReLU(0.2, inplace=True),
            # state size. (ndf*4) x 8 x 8
            nn.Conv2d(ndf * 4, ndf * 8, 4, 2, 1, bias=False),
            nn.BatchNorm2d(ndf * 8),
            nn.LeakyReLU(0.2, inplace=True),
            # state size. (ndf*8) x 4 x 4
            nn.Conv2d(ndf * 8, 1, 4, 1, 0, bias=False),
            nn.Sigmoid()
        )

    def forward(self, input):
        # Data parallelization is handled by the FL framework, not inside the module
        output = self.main(input)
        return output.view(-1, 1).squeeze(1)

def build_model(config: dict) -> nn.Module:
    """
    Builds either a Generator or Discriminator model based on config.
    For federated GAN training, typically only one component (e.g., Discriminator)
    is federated per client, or the server manages a global Generator.
    """
    kwargs = config.get("model_kwargs", {})
    model_type = config.get("model_type", "discriminator").lower() # Default to federating Discriminator

    if model_type == "generator":
        model = Generator(
            nz=kwargs.get("nz", 100),
            ngf=kwargs.get("ngf", 64),
            nc=kwargs.get("nc", 3)
        )
    elif model_type == "discriminator":
        model = Discriminator(
            ndf=kwargs.get("ndf", 64),
            nc=kwargs.get("nc", 3)
        )
    else:
        raise ValueError(f"Unknown model_type: {model_type}. Expected 'generator' or 'discriminator'.")

    # Apply custom weights initialization
    model.apply(weights_init)
    return model


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    """
    Builds a DataLoader for the specified dataset and split.
    """
    local = config.get("local", {})
    batch_size = local.get("batch_size", config.get("batch_size", 64))
    num_workers = local.get("num_workers", config.get("num_workers", 2))
    pin_memory = local.get("pin_memory", True)
    
    dataset_kwargs = config.get("dataset_kwargs", {})
    data_path = config.get("data_path", ".")
    dataset_name = dataset_kwargs.get("dataset", "cifar10").lower()
    image_size = dataset_kwargs.get("image_size", config.get("image_size", 64))
    manual_seed = config.get("seed", 42)

    # Set random seed for dataset splitting
    random.seed(manual_seed)
    torch.manual_seed(manual_seed)

    dataset_params = _get_dataset_params(dataset_name)
    nc = dataset_params['nc']
    normalize_mean = dataset_params['mean']
    normalize_std = dataset_params['std']

    transform_list = [
        transforms.Resize(image_size),
        transforms.CenterCrop(image_size),
        transforms.ToTensor(),
        transforms.Normalize(normalize_mean, normalize_std),
    ]

    # Special handling for MNIST, which might not need CenterCrop if image_size matches
    if dataset_name == 'mnist' and image_size == 28:
        transform_list = [
            transforms.Resize(image_size),
            transforms.ToTensor(),
            transforms.Normalize(normalize_mean, normalize_std),
        ]

    transform = transforms.Compose(transform_list)

    # Dataset loading logic from original script
    if dataset_name in ['imagenet', 'folder', 'lfw']:
        full_dataset = dset.ImageFolder(root=data_path, transform=transform)
    elif dataset_name == 'lsun':
        classes_str = dataset_kwargs.get('classes', 'bedroom')
        classes = [c + '_train' for c in classes_str.split(',')]
        full_dataset = dset.LSUN(root=data_path, classes=classes, transform=transform)
    elif dataset_name == 'cifar10':
        full_dataset = dset.CIFAR10(root=data_path, download=True, transform=transform)
    elif dataset_name == 'mnist':
        full_dataset = dset.MNIST(root=data_path, download=True, transform=transform)
    elif dataset_name == 'fake':
        # Synthetic fallback as in example, image_size passed for FakeData setup
        full_dataset = dset.FakeData(image_size=(nc, image_size, image_size),
                                     transform=transform)
    else:
        raise ValueError(f"Unsupported dataset: {dataset_name}")

    if not full_dataset:
        raise RuntimeError(f"Dataset '{dataset_name}' could not be loaded from '{data_path}'")

    val_ratio = config.get("val_ratio", 0.1) # Default to 10% validation
    n_val = max(1, int(len(full_dataset) * val_ratio)) if val_ratio > 0 else 0
    n_train = len(full_dataset) - n_val

    if n_val > 0:
        train_ds, val_ds = random_split(
            full_dataset, [n_train, n_val],
            generator=torch.Generator().manual_seed(manual_seed),
        )
        ds = train_ds if split == "train" else val_ds
    else:
        # If val_ratio is 0, use the full dataset for training
        ds = full_dataset


    return DataLoader(
        ds,
        batch_size=batch_size,
        shuffle=(split == "train"), # Only shuffle training data
        num_workers=num_workers,
        pin_memory=pin_memory and torch.cuda.is_available(),
    )


def train_step(
    model: nn.Module,
    batch: tuple | list,
    optimizer, # This optimizer is expected to be for `model`
    config: dict,
) -> torch.Tensor:
    """
    Performs one forward pass for the Discriminator (model) and returns its loss.
    This function assumes `model` is the Discriminator and will train it to
    distinguish real images from fake images generated by a *fixed* Generator
    (whose parameters are provided via config or pre-loaded).

    The FL runtime calls loss.backward() and optimizer.step() externally —
    do NOT do either here, and do NOT detach() or .item() the returned loss.
    """
    device = next(model.parameters()).device
    inputs, _ = batch # GAN discriminators typically only need the image data

    # Move inputs to the correct device
    if isinstance(inputs, torch.Tensor):
        real_cpu = inputs.to(device)
    else:
        raise TypeError(f"Unsupported batch input type: {type(inputs)}")

    batch_size = real_cpu.size(0)

    # Labels for real and fake
    real_label = config.get("real_label", 1)
    fake_label = config.get("fake_label", 0)
    
    # Loss criterion
    criterion = nn.BCELoss()

    # --- Discriminator training with real images ---
    # Create labels for real images
    label_real = torch.full((batch_size,), real_label,
                            dtype=real_cpu.dtype, device=device)
    
    # Forward pass real batch through Discriminator
    output_real = model(real_cpu)
    errD_real = criterion(output_real, label_real)


    # --- Discriminator training with fake images ---
    # To generate fake images, we need a Generator.
    # We assume Generator's architecture and parameters are provided via config
    # and it is kept fixed during this Discriminator training step.
    generator_kwargs = config.get("generator_kwargs", {})
    generator_state_dict_path = config.get("generator_state_dict_path")
    
    # Get model parameters for Generator from config if not explicit in kwargs
    gen_nz = generator_kwargs.get("nz", config.get("nz", 100))
    gen_ngf = generator_kwargs.get("ngf", config.get("ngf", 64))
    gen_nc = generator_kwargs.get("nc", config.get("nc", 3))

    # Instantiate and load Generator.
    # The Generator is NOT the `model` parameter, so its weights won't be updated by `optimizer`.
    # It must be `eval()` mode and `torch.no_grad()` is used for its forward pass.
    generator = Generator(nz=gen_nz, ngf=gen_ngf, nc=gen_nc).to(device)
    if generator_state_dict_path:
        if os.path.exists(generator_state_dict_path):
            generator.load_state_dict(torch.load(generator_state_dict_path))
        else:
            # Fallback for synthetic, or if path is invalid/empty for actual data
            print(f"Warning: Generator state dict not found at {generator_state_dict_path}. Using randomly initialized generator for D training.")
    generator.eval() # Ensure Generator is in evaluation mode

    noise_dim = gen_nz
    noise = torch.randn(batch_size, noise_dim, 1, 1, device=device)
    
    # Generate fake image batch with Generator (no gradients for Generator)
    with torch.no_grad():
        fake = generator(noise)

    # Create labels for fake images
    label_fake = torch.full((batch_size,), fake_label,
                            dtype=real_cpu.dtype, device=device)
    
    # Forward pass fake batch through Discriminator
    output_fake = model(fake) # No .detach() needed for `fake` since it's already `with torch.no_grad()`
    errD_fake = criterion(output_fake, label_fake)

    # Combine losses for Discriminator
    errD = errD_real + errD_fake
    
    return errD
