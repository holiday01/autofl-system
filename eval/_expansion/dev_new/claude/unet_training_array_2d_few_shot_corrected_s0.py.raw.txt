"""
Auto-generated FL client module.

Exposes:
  build_model(config)                   -> nn.Module
  build_dataloader(config, split)       -> DataLoader

CONTRACT (read carefully before copying this pattern):

  - Do NOT call loss.backward() inside train_step.
  - Do NOT call optimizer.step() or optimizer.zero_grad() inside train_step.
  - Do NOT call .item() on the returned loss.
  The FL runtime owns backward(), step(), and metric extraction.

import os
from glob import glob

import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader

import monai
from monai.data import ArrayDataset
from monai.transforms import (
    Compose,
    LoadImage,
    RandRotate90,
    RandSpatialCrop,
    ScaleIntensity,
)


# ── Synthetic fallback dataset (no disk I/O) ────────────────────────────

class SyntheticSeg2DDataset(Dataset):
    """In-memory synthetic dataset used when no image files are found on disk."""

    def __init__(self, n: int = 40, image_size: tuple = (1, 128, 128), crop_size: tuple = (96, 96)):
        self.crop_size = crop_size
        c, h, w = image_size
        self.images = torch.randn(n, c, h, w)
        self.segs   = (torch.randn(n, c, h, w) > 0).float()

    def __len__(self):
        return len(self.images)

    def __getitem__(self, idx):
        img = self.images[idx]
        seg = self.segs[idx]
        _, ih, iw = img.shape
        ch, cw = self.crop_size
        top  = torch.randint(0, ih - ch + 1, (1,)).item()
        left = torch.randint(0, iw - cw + 1, (1,)).item()
        return (
            img[:, top:top + ch, left:left + cw],
            seg[:, top:top + ch, left:left + cw],
        )


# ── Transforms ─────────────────────────────────────────────────────────

def _make_train_transform(crop_size: tuple) -> Compose:
    return Compose([
        LoadImage(image_only=True, ensure_channel_first=True),
        ScaleIntensity(),
        RandSpatialCrop(crop_size, random_size=False),
        RandRotate90(prob=0.5, spatial_axes=(0, 1)),
    ])


def _make_val_transform() -> Compose:
    return Compose([
        LoadImage(image_only=True, ensure_channel_first=True),
        ScaleIntensity(),
    ])


# ── FL Interface ────────────────────────────────────────────────────────

def build_model(config: dict) -> nn.Module:
    kwargs = config.get("model_kwargs", {})
    return monai.networks.nets.UNet(
        spatial_dims=kwargs.get("spatial_dims", 2),
        in_channels=kwargs.get("in_channels", 1),
        out_channels=kwargs.get("out_channels", 1),
        channels=tuple(kwargs.get("channels", (16, 32, 64, 128, 256))),
        strides=tuple(kwargs.get("strides", (2, 2, 2, 2))),
        num_res_units=kwargs.get("num_res_units", 2),
    )


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    local       = config.get("local", {})
    batch_size  = local.get("batch_size", config.get("batch_size", 4 if split == "train" else 1))
    num_workers = local.get("num_workers", config.get("num_workers", 4))
    pin_memory  = local.get("pin_memory", True)
    crop_size   = tuple(config.get("crop_size", (96, 96)))
    val_ratio   = config.get("val_ratio", 0.5)
    data_path   = config.get("data_path", ".")

    images = sorted(glob(os.path.join(data_path, "img*.png")))
    segs   = sorted(glob(os.path.join(data_path, "seg*.png")))

    if images and segs:
        n_val   = max(1, int(len(images) * val_ratio))
        n_train = len(images) - n_val
        if split == "train":
            ds = ArrayDataset(
                images[:n_train], _make_train_transform(crop_size),
                segs[:n_train],   _make_train_transform(crop_size),
            )
        else:
            ds = ArrayDataset(
                images[n_train:], _make_val_transform(),
                segs[n_train:],   _make_val_transform(),
            )
    else:
        # synthetic fallback for testing
        n_synth = config.get("n_synthetic", 40)
        n_val   = max(1, int(n_synth * val_ratio))
        n_train = n_synth - n_val
        image_size = tuple(config.get("image_size", (1, 128, 128)))
        ds = SyntheticSeg2DDataset(
            n=n_train if split == "train" else n_val,
            image_size=image_size,
            crop_size=crop_size,
        )

    return DataLoader(
        ds,
        batch_size=batch_size,
        shuffle=(split == "train"),
        num_workers=num_workers,
        pin_memory=pin_memory and torch.cuda.is_available(),
    )


def train_step(
    model: nn.Module,
    batch: tuple | list,
    optimizer,
    config: dict,
) -> torch.Tensor:
    """
    ONE forward pass.  Returns the raw loss tensor WITH grad_fn attached.
    The FL runtime calls loss.backward() and optimizer.step() externally —
    do NOT do either here, and do NOT detach() or .item() the returned loss.
    """
    device = next(model.parameters()).device
    if isinstance(batch, (list, tuple)):
        batch = [b.to(device) if isinstance(b, torch.Tensor) else b for b in batch]
        inputs, targets = batch[0], batch[1]
    elif isinstance(batch, dict):
        batch = {k: v.to(device) if isinstance(v, torch.Tensor) else v
                 for k, v in batch.items()}
        inputs  = batch.get("input", batch.get("x", batch.get("image")))
        targets = batch.get("label", batch.get("y", batch.get("target")))
    else:
        raise TypeError(f"Unsupported batch type: {type(batch)}")

    outputs = model(inputs)
    criterion = monai.losses.DiceLoss(sigmoid=True)
    loss = criterion(outputs, targets)
    return loss
