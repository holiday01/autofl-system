        path = os.path.join(root, filename)

            data = np.loadtxt(path, delimiter="\t")

            x = data[:, 1:].astype(np.float32)
        else:
            n = 200 if file_split == "train" else 50
            x = np.random.randn(n, seq_len).astype(np.float32)

class FCNClassifier(nn.Module):
    """
    Fully Convolutional Network for timeseries classification.
    PyTorch port of the Keras FCN from Fawaz et al. (arXiv:1611.06455).
    Input: (batch, 1, seq_len)  Output: (batch, num_classes)
    """

    def __init__(self, in_channels: int = 1, num_classes: int = 2,
                 filters: int = 64, kernel_size: int = 3):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Conv1d(in_channels, filters, kernel_size, padding="same"),
            nn.BatchNorm1d(filters),
            nn.ReLU(),
            nn.Conv1d(filters, filters, kernel_size, padding="same"),
            nn.BatchNorm1d(filters),
            nn.ReLU(),
            nn.Conv1d(filters, filters, kernel_size, padding="same"),
            nn.BatchNorm1d(filters),
            nn.ReLU(),
        )
        self.classifier = nn.Linear(filters, num_classes)

    def forward(self, x):
        x = self.encoder(x)
        x = x.mean(dim=-1)  # global average pooling over time axis
        return self.classifier(x)
