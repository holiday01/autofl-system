"""
Auto-generated FL client module.
Original script: SSVEPFormer EEG-BCI classification (Keras → PyTorch port).
Reference: Chen et al., Neural Networks 2023 (doi:10.1016/j.neunet.2023.04.045)

Exposes:
  build_model(config)                   -> nn.Module
  build_dataloader(config, split)       -> DataLoader
  train_step(model, batch, opt, config) -> loss tensor (with grad_fn)

CONTRACT (read carefully before copying this pattern):
  - train_step performs ONE forward pass and returns the raw loss tensor.
  - The returned tensor MUST have grad_fn attached (do NOT call .detach()).
  - Do NOT call loss.backward() inside train_step.
  - Do NOT call optimizer.step() or optimizer.zero_grad() inside train_step.
  - Do NOT call .item() on the returned loss.
  The FL runtime owns backward(), step(), and metric extraction.
"""

import os
import numpy as np
import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader, random_split
from scipy.signal import butter, filtfilt
from scipy.io import loadmat


# ── Signal utilities ─────────────────────────────────────────────────────────

def _filter_eeg(data: np.ndarray, fs: int = 256, band=(8, 64), order: int = 4) -> np.ndarray:
    B, A = butter(order, np.array(band) / (fs / 2), btype="bandpass")
    return filtfilt(B, A, data, axis=0)


# ── Dataset ───────────────────────────────────────────────────────────────────

class SSVEPDataset(Dataset):
    """
    Loads EEG trials from Nakanishi-format .mat files (s1.mat … s10.mat).

    Each trial is (n_channels, n_samples); labels are 0-indexed class ids.
    Falls back to synthetic data when the data folder is absent or empty.
    """

    def __init__(
        self,
        data_folder: str,
        subject_ids,
        fs: int = 256,
        duration: float = 1.0,
        band=(8, 64),
        order: int = 4,
        onset: float = 0.135,
        n_classes: int = 12,
    ):
        onset_idx = 38 + int(onset * fs)
        n_samp = int(duration * fs)
        all_trials, all_labels = [], []

        for sid in subject_ids:
            mat_path = os.path.join(data_folder, f"s{sid}.mat")
            if not os.path.exists(mat_path):
                continue
            # original .mat layout → (samples, channels, blocks, targets) after transpose
            eeg = loadmat(mat_path)["eeg"].transpose(2, 1, 3, 0)
            eeg = _filter_eeg(eeg, fs=fs, band=band, order=order)
            eeg = eeg[onset_idx : onset_idx + n_samp]        # (n_samp, C, B, T)
            n_s, n_ch, n_blocks, n_targets = eeg.shape
            # F-order reshape: blocks change fastest → trial order is
            # [all blocks for target 0, all blocks for target 1, …]
            eeg = eeg.reshape(n_s, n_ch, n_blocks * n_targets, order="F")
            eeg = eeg.transpose(2, 1, 0).astype(np.float32)  # (trials, C, n_samp)
            y = np.repeat(np.arange(n_targets), n_blocks).astype(np.int64)
            all_trials.append(eeg)
            all_labels.append(y)

        if all_trials:
            self.X = np.concatenate(all_trials, axis=0)
            self.Y = np.concatenate(all_labels, axis=0)
        else:
            self.X = np.random.randn(180, 8, n_samp).astype(np.float32)
            self.Y = np.random.randint(0, n_classes, 180, dtype=np.int64)

    def __len__(self) -> int:
        return len(self.Y)

    def __getitem__(self, idx):
        return (
            torch.from_numpy(self.X[idx]),
            torch.tensor(self.Y[idx], dtype=torch.long),
        )


# ── Model layers ──────────────────────────────────────────────────────────────

class ComplexSpectrum(nn.Module):
    """Non-trainable FFT layer: concatenates real and imaginary slices of the spectrum."""

    def __init__(self, nfft: int, fft_start: int, fft_end: int):
        super().__init__()
        self.nfft = nfft
        self.fft_start = fft_start
        self.fft_end = fft_end

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        n = x.shape[-1]
        xf = torch.fft.rfft(x, n=self.nfft, dim=-1)
        real = xf.real[..., self.fft_start : self.fft_end] / n
        imag = xf.imag[..., self.fft_start : self.fft_end] / n
        return torch.cat([real, imag], dim=-1)  # (N, C, 2*n_freq)


class ChannelComb(nn.Module):
    def __init__(self, n_channels: int, n_feat: int, drop_rate: float = 0.5):
        super().__init__()
        self.conv = nn.Conv1d(n_channels, 2 * n_channels, kernel_size=1)
        nn.init.normal_(self.conv.weight, 0.0, 0.01)
        nn.init.zeros_(self.conv.bias)
        self.norm = nn.LayerNorm(n_feat)
        self.act = nn.GELU()
        self.drop = nn.Dropout(drop_rate)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.conv(x)
        x = self.norm(x)
        x = self.act(x)
        return self.drop(x)  # (N, 2*C, n_feat)


class ConvAttention(nn.Module):
    def __init__(self, n_channels: int, n_feat: int, drop_rate: float = 0.5):
        super().__init__()
        self.norm = nn.LayerNorm(n_feat)
        self.conv = nn.Conv1d(n_channels, n_channels, kernel_size=31, padding=15)
        nn.init.normal_(self.conv.weight, 0.0, 0.01)
        nn.init.zeros_(self.conv.bias)
        self.act = nn.GELU()
        self.drop = nn.Dropout(drop_rate)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        residual = x
        x = self.norm(x)
        x = self.conv(x)
        x = self.act(x)
        x = self.drop(x)
        return x + residual


class ChannelMLP(nn.Module):
    def __init__(self, n_feat: int, drop_rate: float = 0.5):
        super().__init__()
        self.norm = nn.LayerNorm(n_feat)
        # nn.Linear on (N, C, F) applies independently to the F dim of each channel
        self.linear = nn.Linear(n_feat, n_feat)
        nn.init.normal_(self.linear.weight, 0.0, 0.01)
        nn.init.zeros_(self.linear.bias)
        self.act = nn.GELU()
        self.drop = nn.Dropout(drop_rate)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        residual = x
        x = self.norm(x)
        x = self.linear(x)
        x = self.act(x)
        x = self.drop(x)
        return x + residual


class Encoder(nn.Module):
    def __init__(self, n_channels: int, n_feat: int, drop_rate: float = 0.5):
        super().__init__()
        self.attn1 = ConvAttention(n_channels, n_feat, drop_rate)
        self.mlp1 = ChannelMLP(n_feat, drop_rate)
        self.attn2 = ConvAttention(n_channels, n_feat, drop_rate)
        self.mlp2 = ChannelMLP(n_feat, drop_rate)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.attn1(x)
        x = self.mlp1(x)
        x = self.attn2(x)
        return self.mlp2(x)


class MlpHead(nn.Module):
    def __init__(self, flat_dim: int, n_classes: int, drop_rate: float = 0.5):
        super().__init__()
        hidden = 6 * n_classes
        self.flatten = nn.Flatten()
        self.drop1 = nn.Dropout(drop_rate)
        self.linear1 = nn.Linear(flat_dim, hidden)
        nn.init.normal_(self.linear1.weight, 0.0, 0.01)
        nn.init.zeros_(self.linear1.bias)
        self.norm = nn.LayerNorm(hidden)
        self.act = nn.GELU()
        self.drop2 = nn.Dropout(drop_rate)
        self.linear2 = nn.Linear(hidden, n_classes)
        nn.init.normal_(self.linear2.weight, 0.0, 0.01)
        nn.init.zeros_(self.linear2.bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.flatten(x)
        x = self.drop1(x)
        x = self.linear1(x)
        x = self.norm(x)
        x = self.act(x)
        x = self.drop2(x)
        return self.linear2(x)


class SSVEPFormer(nn.Module):
    """
    PyTorch port of SSVEPFormer (Chen et al., Neural Networks 2023).
    Input : (N, n_channels, n_samples) EEG tensor.
    Output: (N, n_classes) logits — softmax is NOT applied here;
            use nn.CrossEntropyLoss which expects raw logits.
    """

    def __init__(
        self,
        n_channels: int = 8,
        fs: int = 256,
        resolution: float = 0.25,
        fq_band=(8, 64),
        n_classes: int = 12,
        drop_rate: float = 0.5,
    ):
        super().__init__()
        nfft = round(fs / resolution)
        fft_start = int(fq_band[0] / resolution)
        fft_end = int(fq_band[1] / resolution) + 1
        n_freq = fft_end - fft_start
        n_feat = 2 * n_freq        # real + imag concatenated
        n_ch_enc = 2 * n_channels  # after ChannelComb

        self.spectrum = ComplexSpectrum(nfft, fft_start, fft_end)
        self.channel_comb = ChannelComb(n_channels, n_feat, drop_rate)
        self.encoder1 = Encoder(n_ch_enc, n_feat, drop_rate)
        self.encoder2 = Encoder(n_ch_enc, n_feat, drop_rate)
        self.head = MlpHead(n_ch_enc * n_feat, n_classes, drop_rate)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.spectrum(x)
        x = self.channel_comb(x)
        x = self.encoder1(x)
        x = self.encoder2(x)
        return self.head(x)


# ── FL Interface ──────────────────────────────────────────────────────────────

def build_model(config: dict) -> nn.Module:
    kwargs = config.get("model_kwargs", {})
    return SSVEPFormer(**kwargs)


def build_dataloader(config: dict, split: str = "train") -> DataLoader:
    local = config.get("local", {})
    batch_size  = local.get("batch_size", config.get("batch_size", 128))
    num_workers = local.get("num_workers", config.get("num_workers", 2))
    pin_memory  = local.get("pin_memory", True)

    data_path      = config.get("data_path", "./cca_ssvep")
    dataset_kwargs = dict(config.get("dataset_kwargs", {}))

    n_subjects = config.get("n_subjects_total", 10)
    all_ids    = list(range(1, n_subjects + 1))
    train_ids  = config.get("train_subject_ids", all_ids)
    val_ids    = config.get("val_subject_ids", [])

    subject_ids = train_ids if split == "train" else (val_ids or train_ids)

    dataset = SSVEPDataset(
        data_folder=data_path,
        subject_ids=subject_ids,
        **dataset_kwargs,
    )

    # Random val split only when no explicit val subjects are provided
    if split == "train" and not val_ids:
        val_ratio = config.get("val_ratio", 0.1)
        n_val   = max(1, int(len(dataset) * val_ratio))
        n_train = len(dataset) - n_val
        train_ds, _ = random_split(
            dataset, [n_train, n_val],
            generator=torch.Generator().manual_seed(config.get("seed", 42)),
        )
        ds = train_ds
    else:
        ds = dataset

    return DataLoader(
        ds,
        batch_size=batch_size,
        shuffle=(split == "train"),
        num_workers=num_workers,
        pin_memory=pin_memory and torch.cuda.is_available(),
    )


def train_step(
    model: nn.Module,
    batch: tuple | list,
    optimizer,
    config: dict,
) -> torch.Tensor:
    """
    ONE forward pass.  Returns the raw loss tensor WITH grad_fn attached.
    The FL runtime calls loss.backward() and optimizer.step() externally —
    do NOT do either here, and do NOT detach() or .item() the returned loss.
    """
    device = next(model.parameters()).device

    if isinstance(batch, (list, tuple)):
        batch = [b.to(device) if isinstance(b, torch.Tensor) else b for b in batch]
        inputs, targets = batch[0], batch[1]
    elif isinstance(batch, dict):
        batch = {k: v.to(device) if isinstance(v, torch.Tensor) else v
                 for k, v in batch.items()}
        inputs  = batch.get("input", batch.get("x", batch.get("eeg")))
        targets = batch.get("label", batch.get("y", batch.get("target")))
    else:
        raise TypeError(f"Unsupported batch type: {type(batch)}")

    outputs = model(inputs)
    loss = nn.CrossEntropyLoss()(outputs, targets)
    return loss
